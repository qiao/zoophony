// Temporary probe: the light of the key light at the stage points, as a part of the center.
import { Vector3 } from 'three';
import { KEY_LIGHT, createLights } from '../src/scene/lights.ts';
const deg = Number(process.argv[2] ?? 21.6);
(KEY_LIGHT as { angle: number }).angle = (deg * Math.PI) / 180;
const aim = Number(process.argv[3] ?? 0.5);
(KEY_LIGHT as { aimToPodium: number }).aimToPodium = aim;
import { conductorOn } from '../src/models/layout-scene.ts';
import { SpotLight } from 'three';
const side = 11.19;
const conductor = conductorOn([0, 0, side / 2 - 1]);
const g = createLights(side, 'high', conductor, { beam: false });
const spot = g.children.find((c) => c instanceof SpotLight && c.name === 'key') as SpotLight;
const axis = spot.target.position.clone().sub(spot.position);
const cone = (a: number) => { const outer = spot.angle; const inner = outer * (1 - spot.penumbra); if (a >= outer) return 0; if (a <= inner) return 1; const t = (Math.cos(a) - Math.cos(outer)) / (Math.cos(inner) - Math.cos(outer)); return t * t * (3 - 2 * t); };
const light = (x: number, z: number) => { const to = new Vector3(x, 0, z).sub(spot.position); return (cone(to.angleTo(axis)) * -to.y) / to.length() ** 3; };
const c = light(0, 0); const h = side / 2;
for (const [n, x, z] of [['left mid', -h, 0], ['right mid', h, 0], ['front left', -h, h], ['front right', h, h], ['back right', h, -h], ['back left', -h, -h], ['back mid', 0, -h]] as const) process.stdout.write(`${n} ${(light(x, z) / c).toFixed(2)}  `);
console.log(`(angle ${deg}°, aim ${aim})`);
