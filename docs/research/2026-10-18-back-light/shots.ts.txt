// Lists the close shots of a demo whose subject sits in the back rows (radius from the podium
// of 7 m or more), with their times, as the app plans them.
import { buildScore } from '../../src/score/score.ts';
import { layoutScene } from '../../src/models/layout-scene.ts';
import { ModelCache } from '../../src/models/model-cache.ts';
import { SHOT_SEED, prepareShots } from '../../src/shots/prepare.ts';
import { patchedDemos } from '../../tools/patch-demos.ts';
const want = process.argv[2] ?? 'boulanger';
const demo = (await patchedDemos()).find((d) => d.file.includes(want))!;
const result = buildScore(demo.bytes, 'auto');
if (!result.ok) throw new Error('bad');
const models = new ModelCache();
const layout = layoutScene(result.score, models);
const plan = prepareShots(result.score, layout, SHOT_SEED, models);
const [px, , pz] = layout.podium;
for (const s of plan.shots) {
  if (s.seats.length === 0) continue;
  const players = layout.players.filter((p) => s.seats.includes(p.cast.seatIndex));
  const r = Math.max(...players.map((p) => Math.hypot(p.position[0] - px, p.position[2] - pz)));
  const species = [...new Set(players.map((p) => p.cast.species))].join(',');
  if (process.argv[3] !== 'all' && r < 7) continue;
  console.log(`${s.kind} ${s.startSec.toFixed(1)}-${s.endSec.toFixed(1)} s seats ${s.seats.join(',')} ${species} r ${r.toFixed(1)} radius ${s.radius.toFixed(2)}`);
}
