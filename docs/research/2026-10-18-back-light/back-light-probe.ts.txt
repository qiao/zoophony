// Temporary probe: the light on the front of each player, by row, as a part of the first row.
import { SpotLight, Vector3 } from 'three';
import { layoutScene } from '../src/models/layout-scene.ts';
import { ModelCache } from '../src/models/model-cache.ts';
import { createLights } from '../src/scene/lights.ts';
import { defaultAim } from '../src/scene/light-pool.ts';
import { scoreOf } from '../src/score/test-score.ts';
import { patchedDemos } from '../tools/patch-demos.ts';

const CHEST = 0.8;
const want = process.argv[2] ?? 'beethoven-9';
const demo = (await patchedDemos()).find((d) => d.file.includes(want));
if (!demo) throw new Error(`no demo ${want}`);
const layout = layoutScene(scoreOf(demo.bytes), new ModelCache());
const g = createLights(layout.stageSide, 'high', layout.conductor, { beam: false });
const spots = g.children.filter((c): c is SpotLight => c instanceof SpotLight);

function cone(spot: SpotLight, a: number): number {
  const outer = spot.angle;
  const inner = outer * (1 - spot.penumbra);
  if (a >= outer) return 0;
  if (a <= inner) return 1;
  const t = (Math.cos(a) - Math.cos(outer)) / (Math.cos(inner) - Math.cos(outer));
  return t * t * (3 - 2 * t);
}
function light(spot: SpotLight, p: Vector3, n: Vector3): number {
  const axis = spot.target.position.clone().sub(spot.position);
  const to = p.clone().sub(spot.position);
  const lambert = Math.max(0, -to.clone().normalize().dot(n));
  return (spot.intensity * cone(spot, to.angleTo(axis)) * lambert) / to.lengthSq();
}

const [px, , pz] = layout.podium;
const aim = defaultAim(layout.stageSide, layout.conductor ? layout.podium : null);
const pool = (x: number, z: number): number => { const dx = (x - aim.center[0]) / aim.radius[0]; const dz = (z - aim.center[1]) / aim.radius[1]; return aim.floor + (1 - aim.floor) * Math.exp(-(dx * dx + dz * dz)); };
const usePool = process.argv[3] !== 'nopool';
const rows = new Map<number, { key: number; side: number; n: number; species: Set<string> }>();
for (const p of layout.players) {
  const [x, y, z] = p.position;
  const radius = Math.hypot(x - px, z - pz);
  const row = Math.round(radius / 1.2);
  const point = new Vector3(x, y + CHEST, z);
  const front = new Vector3(-Math.sin(p.facing), 0, -Math.cos(p.facing));
  let key = 0;
  let side = 0;
  const f = usePool ? pool(x, z) : 1;
  for (const s of spots) {
    if (s.name === 'key') key += f * light(s, point, front);
    else if (s.name === 'side') side += f * light(s, point, front);
  }
  const r = rows.get(row) ?? { key: 0, side: 0, n: 0, species: new Set() };
  r.key += key;
  r.side += side;
  r.n += 1;
  r.species.add(p.cast.species);
  rows.set(row, r);
}
const sorted = [...rows.entries()].toSorted((a, b) => a[0] - b[0]);
const first = sorted[0]![1];
const base = (first.key + first.side) / first.n;
console.log(`${demo.file}: S ${layout.stageSide.toFixed(2)} m, ${layout.players.length} players`);
for (const [row, r] of sorted) {
  const k = r.key / r.n;
  const s = r.side / r.n;
  console.log(
    `radius ~${(row * 1.2).toFixed(1)} m  n ${r.n}  side ${(s / base).toFixed(2)}  key(front) ${(k / base).toFixed(2)}  total ${((k + s) / base).toFixed(2)}  ${[...r.species].join(',')}`,
  );
}
