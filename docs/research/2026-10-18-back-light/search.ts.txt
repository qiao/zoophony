// Grid search: for each geometry of the back-row lights, the illuminance that gives the back row
// 0.85 of the first row (mean of three demos), and the spill on the first row at that value.
import { SpotLight, Vector3 } from 'three';
import { layoutScene } from '../../src/models/layout-scene.ts';
import { ModelCache } from '../../src/models/model-cache.ts';
import * as lights from '../../src/scene/lights.ts';
import * as pools from '../../src/scene/light-pool.ts';
import { scoreOf } from '../../src/score/test-score.ts';
import { patchedDemos } from '../../tools/patch-demos.ts';

const BACK = lights.BACK_ROW_LIGHT as unknown as Record<string, unknown>;
const POOL = pools.LIGHT_POOL as unknown as Record<string, unknown>;
const demos = await patchedDemos();
const layouts = ['beethoven-9', 'brahms', 'boulanger'].map((w) => {
  const d = demos.find((x) => x.file.includes(w))!;
  return { name: w, layout: layoutScene(scoreOf(d.bytes), new ModelCache()) };
});
function cone(spot: SpotLight, a: number): number {
  const outer = spot.angle, inner = outer * (1 - spot.penumbra);
  if (a >= outer) return 0;
  if (a <= inner) return 1;
  const t = (Math.cos(a) - Math.cos(outer)) / (Math.cos(inner) - Math.cos(outer));
  return t * t * (3 - 2 * t);
}
function light(spot: SpotLight, p: Vector3, n: Vector3): number {
  const axis = spot.target.position.clone().sub(spot.position);
  const to = p.clone().sub(spot.position);
  const lambert = Math.max(0, -to.clone().normalize().dot(n));
  return (spot.intensity * cone(spot, to.angleTo(axis)) * lambert) / to.lengthSq();
}
function rowsOf(layout: (typeof layouts)[number]['layout']) {
  const g = lights.createLights(layout.stageSide, 'high', layout.conductor, { beam: false });
  const spots = g.children.filter((c): c is SpotLight => c instanceof SpotLight);
  const [px, , pz] = layout.podium;
  const aim = pools.defaultAim(layout.stageSide, layout.podium);
  const pool = (x: number, z: number) => {
    const dx = (x - aim.center[0]) / aim.radius[0];
    const dz = (z - aim.center[1]) / (z < aim.center[1] ? aim.reachBack : aim.radius[1]);
    return aim.floor + (1 - aim.floor) * Math.exp(-(dx * dx + dz * dz));
  };
  const rows = new Map<number, { main: number[]; back: number[]; choir: boolean }>();
  for (const p of layout.players) {
    const [x, y, z] = p.position;
    const row = Math.round(Math.hypot(x - px, z - pz) / 1.2);
    const point = new Vector3(x, y + 0.8, z);
    const front = new Vector3(-Math.sin(p.facing), 0, -Math.cos(p.facing));
    const f = pool(x, z);
    let main = 0, back = 0;
    for (const s of spots) {
      if (s.name === 'back-row') back += f * light(s, point, front);
      else main += f * light(s, point, front);
    }
    const r = rows.get(row) ?? { main: [], back: [], choir: p.cast.species === 'penguin' };
    r.main.push(main); r.back.push(back);
    rows.set(row, r);
  }
  // The floor at the back: points of the stage top behind the last row, with the round pool.
  const stagePool = (x: number, z: number) => {
    const dx = (x - aim.center[0]) / aim.radius[0];
    const dz = (z - aim.center[1]) / aim.radius[1];
    return aim.floor + (1 - aim.floor) * Math.exp(-(dx * dx + dz * dz));
  };
  const lastZ = Math.min(...layout.players.map((p) => p.position[2]));
  const half = layout.stageSide / 2;
  let floorMain = 0, floorBack = 0;
  const up = new Vector3(0, 1, 0);
  for (let x = -0.4 * layout.stageSide; x <= 0.4 * layout.stageSide; x += 0.5) {
    for (let z = -half + 0.2; z < lastZ - 0.6; z += 0.25) {
      const point = new Vector3(x, 0, z);
      const f = stagePool(x, z);
      for (const s of spots) {
        if (s.name === 'back-row') floorBack += f * light(s, point, up);
        else if (s.name === 'key') floorMain += f * light(s, point, up);
      }
    }
  }
  const out = [...rows.entries()].toSorted((a, b) => a[0] - b[0]).map(([, r]) => r);
  return Object.assign(out, { floorMain, floorBack });
}
const mean = (v: number[]) => v.reduce((a, b) => a + b, 0) / v.length;
const goal = Number(process.argv[3] ?? 0.85);
const grid = JSON.parse(process.argv[2]!) as Record<string, number[]>;
const keys = Object.keys(grid);
const combos: Record<string, number>[] = [{}];
for (const k of keys) {
  const next: Record<string, number>[] = [];
  for (const c of combos) for (const v of grid[k]!) next.push({ ...c, [k]: v });
  combos.splice(0, combos.length, ...next);
}
const results: { c: Record<string, number>; I: number; spill: number; text: string }[] = [];
for (const c of combos) {
  BACK['illuminance'] = 1;
  const pos = [c['px'] ?? 1.2, c['py'] ?? 1.7, c['pz'] ?? 0.9];
  const tgt = [c['tx'] ?? -0.25, c['ty'] ?? 0, c['tz'] ?? -0.3];
  const floors: number[] = [];
  BACK['position'] = pos; BACK['target'] = tgt;
  BACK['angle'] = ((c['angle'] ?? 14) * Math.PI) / 180;
  BACK['penumbra'] = c['pen'] ?? 0.5;
  if (c['reach'] !== undefined) POOL['reachBack'] = c['reach'];
  // Per demo: first row main mf, unit back uf; back row main mb, unit ub.
  const per = layouts.map(({ layout }) => {
    const rows = rowsOf(layout);
    const first = rows[0]!;
    const backRow = rows.filter((r) => !r.choir).at(-1)!;
    const choir = rows.find((r) => r.choir);
    floors.push(rows.floorBack / rows.floorMain);
    return { mf: mean(first.main), uf: mean(first.back), mb: mean(backRow.main), ub: mean(backRow.back),
      second: rows.filter((r) => !r.choir).at(-2)!, backRow, choir };
  });
  // Solve the mean ratio for I by bisection.
  const ratio = (I: number) => mean(per.map((p) => (p.mb + I * p.ub) / (p.mf + I * p.uf)));
  let lo = 0, hi = 50;
  if (ratio(hi) < goal) { continue; }
  for (let i = 0; i < 50; i++) { const m = (lo + hi) / 2; if (ratio(m) < goal) lo = m; else hi = m; }
  const I = hi;
  const spill = Math.max(...per.map((p) => (I * p.uf) / p.mf));
  const text = per.map((p, i) => {
    const base = p.mf + I * p.uf;
    const r = (row: { main: number[]; back: number[] }) => mean(row.main.map((m, j) => m + I * row.back[j]!)) / base;
    const mn = (row: { main: number[]; back: number[] }) => Math.min(...row.main.map((m, j) => m + I * row.back[j]!)) / base;
    return `${layouts[i]!.name.slice(0, 5)} spill ${((I * p.uf / p.mf) * 100).toFixed(1)}% 2nd ${r(p.second).toFixed(2)} back ${r(p.backRow).toFixed(2)} (min ${mn(p.backRow).toFixed(2)})${p.choir ? ` choir ${r(p.choir).toFixed(2)}` : ''}`;
  }).join(' | ');
  const cost = per.reduce((acc, p) => {
    const base = p.mf + I * p.uf;
    const r = (row: { main: number[]; back: number[] }) => mean(row.main.map((m, j) => m + I * row.back[j]!)) / base;
    const mn = (row: { main: number[]; back: number[] }) => Math.min(...row.main.map((m, j) => m + I * row.back[j]!)) / base;
    return acc + Math.abs(r(p.backRow) - goal) + Math.max(0, r(p.second) - 1) + Math.max(0, goal - 0.25 - mn(p.backRow)) + (p.choir ? Math.abs(r(p.choir) - goal) : 0);
  }, 0) + 10 * Math.max(0, spill - 0.03);
  const floorAdd = Math.max(...floors.map((f) => f * I));
  results.push({ c, I, spill: cost + Number(process.argv[5] ?? 1) * floorAdd, text: `spill ${(spill * 100).toFixed(1)}% back floor +${(floorAdd * 100).toFixed(0)}% of key | ` + text });
}
results.sort((a, b) => a.spill - b.spill);
for (const r of results.slice(0, Number(process.argv[4] ?? 15))) {
  console.log(`${JSON.stringify(r.c)} I ${r.I.toFixed(2)} cost ${r.spill.toFixed(2)}\n   ${r.text}`);
}
