// Renders Cinematic frames of the app on a dev server with the look override of the round
// back-light: it plays a demo, seeks during play, and saves the frame at a song time, with no
// "Reset view", so the shot planner holds the camera (design 5.6).
// Usage: node shot-frames.ts <out dir> <port> <demo> <time s> <name=JSON>... [--solo <seat>]
import { mkdir, writeFile } from 'node:fs/promises';
import { join } from 'node:path';

import { chromium } from '@playwright/test';

import { FIRST_PLAY, SAMPLES_READY } from '../../src/app/dev-hook.ts';
import { chromiumArgs } from '../../tools/chromium-args.ts';
import { openShot } from '../../tools/screenshots.ts';

const [out, port, demo, timeText, ...rest] = process.argv.slice(2);
if (!out || !port || !demo || !timeText) {
  throw new Error('Usage: node shot-frames.ts <out> <port> <demo> <time> <name=JSON>... [--solo <seat>]');
}
const soloAt = rest.indexOf('--solo');
const solo = soloAt >= 0 ? rest[soloAt + 1] : undefined;
const variants = rest.filter((arg, i) => arg.includes('=') && (soloAt < 0 || i !== soloAt + 1));
const time = Number(timeText);
await mkdir(out, { recursive: true });
const browser = await chromium.launch({ args: chromiumArgs(process.platform) });
try {
  for (const variant of variants) {
    const [name, json] = [variant.slice(0, variant.indexOf('=')), variant.slice(variant.indexOf('=') + 1)];
    const query = new URLSearchParams({ look: json });
    if (solo !== undefined) {
      query.set('solo', solo);
    }
    const url = `http://localhost:${port}/?${query.toString()}#demo=${demo}`;
    const page = await openShot(browser, url, {
      file: '',
      path: `/#demo=${demo}`,
      width: 1440,
      height: 900,
      ready: SAMPLES_READY,
    });
    try {
      await page.click(FIRST_PLAY);
      await page.waitForTimeout(1500);
      await page.click('.close-drawer');
      const whole = Math.floor(time);
      // The pointer leaves the bar before the seek, so the bar hides 3 s later, before the time.
      await page.mouse.move(720, 450);
      await page.$eval(
        '.seek',
        (element, value) => {
          if (element instanceof HTMLInputElement) {
            element.value = String(value);
            element.dispatchEvent(new Event('input', { bubbles: true }));
            element.dispatchEvent(new Event('change', { bubbles: true }));
          }
        },
        Math.max(0, whole - 4),
      );
      await page.waitForFunction(
        (sec) => {
          const seek = document.querySelector('.seek');
          return seek instanceof HTMLInputElement && Number(seek.value) >= sec;
        },
        whole,
        { polling: 20, timeout: 60_000 },
      );
      await page.waitForTimeout((time - whole) * 1000);
      await page.waitForSelector('.bar.is-hidden', { state: 'attached', timeout: 10_000 });
      await page.waitForTimeout(300);
      const clock = await page.$eval('.bar .time', (element) => element.textContent ?? '');
      const file = join(out, `${name}.png`);
      await writeFile(file, await page.screenshot({ type: 'png' }));
      console.log(`${file}: song time ${clock}`);
    } finally {
      await page.close();
    }
  }
} finally {
  await browser.close();
}
