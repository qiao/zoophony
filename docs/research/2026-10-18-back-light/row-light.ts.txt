// Probe of the round back-light: the direct light on the front of each player (chest height,
// normal = the facing of the player), by row, as a part of the first row. It is the probe
// back-light-probe.ts of the manager with the back-row lights, the reach of the pool on the
// players (`LightAim.reachBack`), and overrides of the values.
// Usage: node row-light.ts <demo> [nopool] [hot] [<JSON of path -> value>]
//   nopool: no pool factor. hot: the hot spot of the key light on each spotlight with a shadow
//   (the key light and the side lights of the high tier). JSON: such as
//   '{"BACK_ROW_LIGHT.illuminance": 3, "LIGHT_POOL.reachBack": 0.9}'.
import { SpotLight, Vector3 } from 'three';
import { layoutScene } from '../../src/models/layout-scene.ts';
import { ModelCache } from '../../src/models/model-cache.ts';
import * as lights from '../../src/scene/lights.ts';
import * as pools from '../../src/scene/light-pool.ts';
import { scoreOf } from '../../src/score/test-score.ts';
import { patchedDemos } from '../../tools/patch-demos.ts';

const args = process.argv.slice(2);
const want = args[0] ?? 'beethoven-9';
const usePool = !args.includes('nopool');
const useHot = args.includes('hot');
const json = args.find((a) => a.startsWith('{'));
const roots: Record<string, Record<string, unknown>> = {
  BACK_ROW_LIGHT: (lights as Record<string, unknown>)['BACK_ROW_LIGHT'] as Record<string, unknown>,
  SIDE_LIGHT: lights.SIDE_LIGHT as unknown as Record<string, unknown>,
  LIGHT_POOL: pools.LIGHT_POOL as unknown as Record<string, unknown>,
};
if (json) {
  for (const [path, value] of Object.entries(JSON.parse(json) as Record<string, unknown>)) {
    const [root, key] = path.split('.');
    const node = roots[root!];
    if (!node) {
      continue; // the base tree has no such value
    }
    node[key!] = key === 'angle' ? ((value as number) * Math.PI) / 180 : value;
  }
}

const CHEST = 0.8;
const demo = (await patchedDemos()).find((d) => d.file.includes(want));
if (!demo) throw new Error(`no demo ${want}`);
const layout = layoutScene(scoreOf(demo.bytes), new ModelCache());
const g = lights.createLights(layout.stageSide, 'high', layout.conductor, { beam: false });
const spots = g.children.filter((c): c is SpotLight => c instanceof SpotLight);

function cone(spot: SpotLight, a: number): number {
  const outer = spot.angle;
  const inner = outer * (1 - spot.penumbra);
  if (a >= outer) return 0;
  if (a <= inner) return 1;
  const t = (Math.cos(a) - Math.cos(outer)) / (Math.cos(inner) - Math.cos(outer));
  return t * t * (3 - 2 * t);
}
function light(spot: SpotLight, p: Vector3, n: Vector3): number {
  const axis = spot.target.position.clone().sub(spot.position);
  const to = p.clone().sub(spot.position);
  const lambert = Math.max(0, -to.clone().normalize().dot(n));
  return (spot.intensity * cone(spot, to.angleTo(axis)) * lambert) / to.lengthSq();
}

const [px, , pz] = layout.podium;
const aim = pools.defaultAim(layout.stageSide, layout.conductor ? layout.podium : null);
const reachBack = (aim as { reachBack?: number }).reachBack ?? aim.radius[1];
function pool(x: number, z: number): number {
  const dx = (x - aim.center[0]) / aim.radius[0];
  const dz = (z - aim.center[1]) / (z < aim.center[1] ? reachBack : aim.radius[1]);
  return aim.floor + (1 - aim.floor) * Math.exp(-(dx * dx + dz * dz));
}
function hot(x: number, z: number): number {
  const dx = (x - aim.spot[0]) / aim.spotRadius[0];
  const dz = (z - aim.spot[1]) / aim.spotRadius[1];
  return 1 + aim.boost * Math.exp(-(dx * dx + dz * dz));
}

type Row = { key: number; side: number; back: number; n: number; species: Set<string> };
const rows = new Map<number, Row>();
for (const p of layout.players) {
  const [x, y, z] = p.position;
  const row = Math.round(Math.hypot(x - px, z - pz) / 1.2);
  const point = new Vector3(x, y + CHEST, z);
  const front = new Vector3(-Math.sin(p.facing), 0, -Math.cos(p.facing));
  const f = usePool ? pool(x, z) : 1;
  const h = useHot ? hot(x, z) : 1;
  const r = rows.get(row) ?? { key: 0, side: 0, back: 0, n: 0, species: new Set<string>() };
  for (const s of spots) {
    if (s.name === 'key') r.key += f * h * light(s, point, front);
    else if (s.name === 'side') r.side += f * h * light(s, point, front);
    else if (s.name === 'back-row') r.back += f * light(s, point, front);
  }
  r.n += 1;
  r.species.add(p.cast.species);
  rows.set(row, r);
}
const sorted = [...rows.entries()].toSorted((a, b) => a[0] - b[0]);
const first = sorted[0]![1];
const base = (first.key + first.side + first.back) / first.n;
console.log(
  `${demo.file}: S ${layout.stageSide.toFixed(2)} m, ${layout.players.length} players, first row ${base.toFixed(3)} lux, pool ${usePool}, hot ${useHot}`,
);
for (const [row, r] of sorted) {
  const [k, s, b] = [r.key / r.n, r.side / r.n, r.back / r.n];
  console.log(
    `radius ~${(row * 1.2).toFixed(1)} m  n ${String(r.n).padStart(2)}  side ${(s / base).toFixed(2)}  back-row ${(b / base).toFixed(2)}  key ${(k / base).toFixed(2)}  total ${((k + s + b) / base).toFixed(2)}  ${((k + s + b)).toFixed(3)} lux  ${[...r.species].join(',')}`,
  );
}
